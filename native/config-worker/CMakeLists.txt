cmake_minimum_required(VERSION 3.31)
project(OrcaConfigWorker LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
set(ORCA_NATIVE_CACHE "${CMAKE_CURRENT_SOURCE_DIR}/../../.native-cache" CACHE PATH "Pinned source/dependency cache")
set(ORCA_SOURCE_DIR "${ORCA_NATIVE_CACHE}/upstream/repository" CACHE PATH "Pinned Orca source")
set(ORCA_NATIVE_PREFIX "${ORCA_NATIVE_CACHE}/prefix" CACHE PATH "Pinned built dependencies")
set(ORCA_NATIVE_DEPENDENCIES "${ORCA_NATIVE_CACHE}/dependencies" CACHE PATH "Pinned dependency sources")
set(ORCA "${ORCA_SOURCE_DIR}")
set(SRC "${ORCA}/src/libslic3r")
set(EIGEN "${ORCA_NATIVE_DEPENDENCIES}/eigen/eigen-5.0.1")
set(CGAL_DIR "${ORCA_NATIVE_DEPENDENCIES}/cgal/CGAL-5.6.3")
set(BOOST_ROOT "${ORCA_NATIVE_PREFIX}")
set(Boost_NO_SYSTEM_PATHS ON)
set(Boost_NO_BOOST_CMAKE ON)
list(PREPEND CMAKE_PREFIX_PATH "${ORCA_NATIVE_PREFIX}")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${ROOT}/scripts/prepare-sources.py" "${ORCA}" "${CMAKE_CURRENT_BINARY_DIR}" RESULT_VARIABLE preparation_result)
if(NOT preparation_result EQUAL 0)
 message(FATAL_ERROR "Pinned native configuration source verification failed")
endif()
find_package(CGAL REQUIRED)
find_package(TBB REQUIRED)
find_package(OpenSSL REQUIRED)
set(Boost_USE_STATIC_LIBS ON)
find_package(Boost 1.84.0 EXACT REQUIRED COMPONENTS log filesystem thread chrono atomic regex date_time)
add_executable(orca-config-worker worker/main.cpp "${CMAKE_CURRENT_BINARY_DIR}/native-preset.cpp" "${CMAKE_CURRENT_BINARY_DIR}/native-thumbnail-config.cpp" ${ORCA}/deps_src/semver/semver.c
 ${SRC}/PlaceholderParser.cpp "${CMAKE_CURRENT_BINARY_DIR}/native-compatibility-flow.cpp"
 ${SRC}/Preset.cpp ${SRC}/libslic3r.cpp ${SRC}/Config.cpp ${SRC}/PrintConfig.cpp ${SRC}/MaterialType.cpp
 ${SRC}/LocalesUtils.cpp ${SRC}/Utils.cpp ${SRC}/Semver.cpp
 ${SRC}/Point.cpp ${SRC}/Polygon.cpp ${SRC}/Polyline.cpp ${SRC}/MultiPoint.cpp
 ${SRC}/Line.cpp ${SRC}/ExPolygon.cpp ${SRC}/BoundingBox.cpp ${SRC}/ClipperUtils.cpp ${SRC}/clipper.cpp)
target_include_directories(orca-config-worker PRIVATE "${ROOT}/generated" "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/native" "${SRC}/GCode" "${ORCA}/src" "${SRC}" "${ORCA}/deps_src" "${ORCA}/deps_src/admesh" "${ORCA}/deps_src/clipper" "${ORCA}/deps_src/clipper2/Clipper2Lib/include" "${ORCA}/deps_src/fast_float" "${ORCA}/deps_src/libigl" "${ORCA}/deps_src/qhull/src" "${EIGEN}" "${ORCA_NATIVE_PREFIX}/include" "${ORCA_NATIVE_DEPENDENCIES}/cereal/cereal-1.3.0/include")
target_compile_definitions(orca-config-worker PRIVATE NDEBUG USE_TBB TBB_USE_CAPTURED_EXCEPTION=0)
target_compile_options(orca-config-worker PRIVATE -O2 -ffunction-sections -fdata-sections)
if(APPLE)
 target_link_options(orca-config-worker PRIVATE -Wl,-dead_strip -Wl,-dead_strip_dylibs)
else()
 target_link_options(orca-config-worker PRIVATE -Wl,--gc-sections)
endif()
target_link_libraries(orca-config-worker PRIVATE CGAL::CGAL TBB::tbb TBB::tbbmalloc OpenSSL::Crypto Boost::log Boost::filesystem Boost::thread Boost::chrono Boost::atomic Boost::regex Boost::date_time)

set_source_files_properties(${SRC}/LocalesUtils.cpp PROPERTIES COMPILE_FLAGS "-include sstream")
