cmake_minimum_required(VERSION 3.31)
project(OrcaEmbossWorker LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ORCA_NATIVE_CACHE "${CMAKE_CURRENT_SOURCE_DIR}/../../.native-cache" CACHE PATH "Pinned source/dependency cache")
set(ORCA_SOURCE_DIR "${ORCA_NATIVE_CACHE}/upstream/repository" CACHE PATH "Pinned Orca source")
set(ORCA_NATIVE_PREFIX "${ORCA_NATIVE_CACHE}/prefix" CACHE PATH "Pinned built dependencies")
set(ORCA_NATIVE_DEPENDENCIES "${ORCA_NATIVE_CACHE}/dependencies" CACHE PATH "Pinned dependency sources")
set(ORCA "${ORCA_SOURCE_DIR}")
set(SRC "${ORCA}/src/libslic3r")
set(EIGEN "${ORCA_NATIVE_DEPENDENCIES}/eigen/eigen-5.0.1")
set(CGAL_DIR "${ORCA_NATIVE_DEPENDENCIES}/cgal/CGAL-5.6.3")
set(BOOST_ROOT "${ORCA_NATIVE_PREFIX}")
set(Boost_NO_SYSTEM_PATHS ON)
set(Boost_NO_BOOST_CMAKE ON)
list(PREPEND CMAKE_PREFIX_PATH "${ORCA_NATIVE_PREFIX}")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/extract-native-helpers.py" "${ORCA}" "${CMAKE_CURRENT_BINARY_DIR}" RESULT_VARIABLE extraction_result)
if(NOT extraction_result EQUAL 0)
 message(FATAL_ERROR "Native helper source verification/extraction failed")
endif()
find_package(CGAL REQUIRED)
find_package(TBB REQUIRED)
find_package(OpenSSL REQUIRED)
set(Boost_USE_STATIC_LIBS ON)
find_package(Boost 1.84.0 EXACT REQUIRED COMPONENTS log filesystem thread chrono atomic regex date_time)
add_executable(orca-emboss-worker worker/main.cpp worker/per-glyph.cpp worker/svg-shape.cpp "${CMAKE_CURRENT_BINARY_DIR}/native-mesh-helpers.cpp"
 ${SRC}/NSVGUtils.cpp ${SRC}/TriangleMeshSlicer.cpp ${SRC}/CutSurface.cpp ${SRC}/Emboss.cpp ${SRC}/Triangulation.cpp ${SRC}/IntersectionPoints.cpp ${SRC}/ExPolygonsIndex.cpp
 ${SRC}/Point.cpp ${SRC}/Polygon.cpp ${SRC}/Polyline.cpp ${SRC}/MultiPoint.cpp
 ${SRC}/Line.cpp ${SRC}/ExPolygon.cpp ${SRC}/BoundingBox.cpp ${SRC}/ClipperUtils.cpp ${SRC}/clipper.cpp)
target_include_directories(orca-emboss-worker PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/generated" "${ORCA}/src" "${SRC}" "${ORCA}/deps_src" "${ORCA}/deps_src/admesh" "${ORCA}/deps_src/clipper" "${ORCA}/deps_src/clipper2/Clipper2Lib/include" "${ORCA}/deps_src/fast_float" "${ORCA}/deps_src/libigl" "${EIGEN}" "${ORCA_NATIVE_PREFIX}/include" "${ORCA_NATIVE_DEPENDENCIES}/cereal/cereal-1.3.0/include")
target_compile_definitions(orca-emboss-worker PRIVATE NDEBUG USE_TBB TBB_USE_CAPTURED_EXCEPTION=0)
target_compile_options(orca-emboss-worker PRIVATE -O2 -ffunction-sections -fdata-sections)
if(APPLE)
 target_link_options(orca-emboss-worker PRIVATE -Wl,-dead_strip -Wl,-dead_strip_dylibs)
else()
 target_link_options(orca-emboss-worker PRIVATE -Wl,--gc-sections)
endif()
target_link_libraries(orca-emboss-worker PRIVATE CGAL::CGAL TBB::tbb TBB::tbbmalloc OpenSSL::Crypto Boost::log Boost::filesystem Boost::thread Boost::chrono Boost::atomic Boost::regex Boost::date_time)
